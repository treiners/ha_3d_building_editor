import { buildInspectionModel } from "../src/interaction/InspectionModel.js";
import { renderEditableInspectionHtml } from "../src/interaction/EditableInspectionDocument.js";
import { openingTargetStyles, renderOpeningClickTargets } from "../src/interaction/OpeningClickTargets.js";
import type { Connector } from "../src/model/Connector.js";
import type { Room } from "../src/model/Room.js";
import type { WindowOpening } from "../src/model/WindowOpening.js";
import { renderFloorSvg } from "../src/svg/SvgRenderer.js";
import { enableRoomAndWallInspection } from "../src/svg/SvgRendererInspectionPatch.js";
import { resolveConnectors }
  from "../src/generation/ConnectorResolver.js";

import { resolveWallOpenings }
  from "../src/generation/OpeningResolver.js";

import { resolveWindows }
  from "../src/generation/WindowResolver.js";

import { addArchitecturalSymbols }
  from "../src/svg/SvgArchitecturalSymbolsPatch.js";

const generated =
  generateFloorGeometry(source.rooms);

const connectors =
  resolveConnectors(
    source.connectors ?? [],
    generated,
  );

const opened =
  resolveWallOpenings(
    generated,
    connectors,
  );

const visibleFloor = {
  ...generated,
  wallSegments: opened.wallSegments,
};

const windows =
  resolveWindows(
    (source.openings ?? []).filter(
      (opening) =>
        opening.type === "window",
    ),
    generated,
  );

interface BuildingFile { building:{name?:string;floors:Array<{rooms:Room[];connectors?:Connector[];openings?:WindowOpening[]}>} }
const input=process.argv[2]??"examples/reference-apartment-v1.2.json";
const output=process.argv[3]??"examples/reference-apartment-openings-editor.html";
const document=JSON.parse(readFileSync(input,"utf8")) as BuildingFile;const source=document.building.floors[0];if(!source)throw new Error("No floor found.");
const floor=generateFloorGeometry(source.rooms);const windows=(source.openings??[]).filter(o=>o.type==="window");let svg=renderFloorSvg(source.rooms,visibleFloor,{title:document.building.name??"Reference Apartment"}); svg = enableRoomAndWallInspection(svg);
const all=source.rooms.flatMap(r=>r.geometry.shape);const minX=Math.min(...all.map((point)=>point[0])),minY=Math.min(...all.map(p=>p[1]));const scale=80,padding=32;
const targets=renderOpeningClickTargets(floor,source.connectors??[],windows,{scale,x:v=>(v-minX)*scale+padding,y:v=>(v-minY)*scale+padding});
svg=svg.replace("</style>",`${openingTargetStyles}\n</style>`).replace('<g class="labels">',`<g class="opening-targets">${targets}</g>\n<g class="labels">`);
const model=buildInspectionModel(source.rooms,floor,source.connectors??[],windows);writeFileSync(output,renderEditableInspectionHtml(svg,model,document,{title:document.building.name??"Openings Editor"}),"utf8");console.log(`Generated ${output}.`);
